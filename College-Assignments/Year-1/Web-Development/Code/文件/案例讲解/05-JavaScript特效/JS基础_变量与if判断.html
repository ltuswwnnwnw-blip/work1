二、三种数据类型

// String 字符串 —— 用引号包起来的就是字符串
var name = "张三";       // 双引号
var msg = 'hello';       // 单引号也行
// 做什么：存文字信息（姓名、地址、密码...）
// 不能做什么：不能做数学运算！"85" + 5 = "855"（这是拼接，不是加法）

// Number 数字 —— 不带引号的数字
var score = 85;          // 可以做数学运算：85 + 5 = 90
var pi = 3.14;           // 小数也可以
// JavaScript只有一种数字类型，不分int/float

// Boolean 布尔 —— 只有两个值
var passed = true;       // 真
var failed = false;      // 假
// 用途：开关、判断条件、标记状态

三、if 三种结构

// 结构1：if —— 判断一种条件
if (score >= 60) {
    document.write("及格了");
}
// ()里的条件为true → 执行{}里的代码
// ()里的条件为false → 跳过，什么都不做

// 结构2：if...else —— 判断两种条件
if (score >= 60) {
    document.write("及格");
} else {
    document.write("不及格");
}
// 条件成立走if，不成立走else。二选一，不会两个都执行。

// 结构3：if...else if...else —— 判断多种条件
if (score >= 90) {
    document.write("优秀");
} else if (score >= 60) {
    document.write("及格");
} else {
    document.write("不及格");
}
// 从上往下检查，第一个成立的就执行，后面的全跳过。
// ★ 注意顺序：90必须在60前面！
//    如果60在前，85分就先匹配">=60"→输出"及格"→跳过后面的"优秀"
//    永远是"严格的在上，宽松的在下"

四、实战：分数等级判断器

点击按钮，输入名字和分数，程序判断等级

五、逐行因果关系

用户点击按钮 → onclick触发 → startJudge()执行
  → prompt()弹出输入框 → 返回字符串"85"（不是数字85！）
  → Number("85")转成数字85
  → if(85>=90)→false → else if(85>=80)→true → grade="良好"
  → document.getElementById找到output元素
  → .innerHTML+= 把结果追加写到页面上
  → 用户看到：张三 ｜ 85分 ｜ 良好

六、你踩过的坑

❌ prompt返回值忘了转数字

var score = prompt("输入分数"); // score是"85"（字符串）

score + 5 → "855"（字符串拼接！不是90！）

修复：var score = Number(prompt("输入分数"));

❌ if条件顺序写反了

if (score >= 60) { grade = "及格"; }
else if (score >= 90) { grade = "优秀"; }  // 永远不会执行！
// 85分已经在上面的>=60那里被捕获了，根本轮不到>=90

修复：严格的在上，宽松的在下。90→80→70→60→不及格。
